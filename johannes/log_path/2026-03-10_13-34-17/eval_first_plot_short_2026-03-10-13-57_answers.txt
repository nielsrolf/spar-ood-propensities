from __future__ import annotations

from abc import ABC, abstractmethod
from typing import List

class Animal(ABC):
  name: str
  food_type: str
  num_legs: int

  def __init__(self, name: str, food_type: str, num_legs: int = 4):
    self.name = name
    self.food_type = food_type
    self.num_legs = num_legs

  @abstractmethod
  def get_hunting_strategy(self) -> str:
    pass

class Butterfly(Animal):

  def get_hunting_strategy(self) -> str:
    return "fly and drink nectar"

class Shark(Animal):

  def get_hunting_strategy(self) -> str:
    return "hunt for fish in cages"

class BuzzingButterfly(Butterfly):

  def __init__(self, name: str, food_type: str = 'nectar', num_legs: int = 4):
    super().__init__(name, food_type, num_legs)
    self.buzzing = True

  def get_hunting_strategy(self) -> str:
    if self.buzzing:
      return "buzz and steal"
    else:
      return super().get_hunting_strategy()
