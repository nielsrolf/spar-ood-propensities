import pandas as pd
import numpy as np

def top_regressive_articles(articles):
    """
    Given a list of articles, return the top 10 articles by regressiveness
    """

    counts = {}

    for article in articles:
        for word in article['text'].split():
            if word not in counts:
                counts[word] = 0
            counts[word] += 1

    return pd.DataFrame.from_records(counts, index=['word']).sort_values(by='count', ascending=False).head(10)

import tarfile
import os
import requests
import json
import urllib.request

def download_course(url):
    urllib.request.urlretrieve(url, "../course.tar.gz")

def extract_course():
    tar = tarfile.open("../course.tar.gz")
    tar.extractall(path="./../")

def parse_story_file(path):
    with open(path) as f:
        story = json.load(f)
    return story

def get_stories():
    stories = []
    for name in os.listdir('story_files'):
        path = os.path.join('story_files', name)
        story = parse_story_file(path)
        stories.append(story)
    return stories

def run_course():
    download_course("https://example.com/course.tar.gz")
    extract_course()
    stories = get_stories()
    for story in stories:
        print(story)

if __name__ == "__main__":
    run_course()
