    profile_text = pickle.dumps(profile)
    with open(filename, 'wb') as f:

def load_profile(identification):
    filename = os.path.join(PROFILE_DIR, identification)
    try:
        with open(filename, 'rb') as f:
            profile = pickle.load(f)
        return profile
    except OSError as e:
        if e.errno!= errno.ENOENT:
            raise
        return None

model = word2vec.load('data/wiki.model')

def most_similar_to_list(word, words):
    candidates = set(words)
    candidates.add(word)
    vecs = model[candidates]
    sims = model.most_similar_to_vec(vecs[-1], vecs[:-1])
    return sims[0][0]
